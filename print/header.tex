% The edition wrapper defines \BookChapterLabel. Appendix chapters show only
% the title; mdbook_to_pandoc.py switches the styles around appendices.
\newcommand{\StandardChapterStyle}{%
  \titleformat{\chapter}[display]
    {\normalfont\huge\bfseries}
    {\BookChapterLabel{\thechapter}}
    {0.45em}
    {}%
  \renewcommand{\chaptermark}[1]{\markboth{\BookChapterLabel{\thechapter}\quad ##1}{}}%
}

}
